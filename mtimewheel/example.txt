========== 批量添加任务 ==========

func main() {
    tw := mtimewheel.New()
    tw.Start()
    defer tw.Stop()

    var wg sync.WaitGroup

    // 批量添加1000个任务
    for i := 0; i < 1000; i++ {
        wg.Add(1)
        taskID := fmt.Sprintf("task-%d", i)
        delay := time.Duration(i%10+1) * time.Second

        tw.AddTask(taskID, delay, func(ctx context.Context) error {
            defer wg.Done()
            fmt.Printf("Task %s executed\n", taskID)
            return nil
        })
    }

    // 等待所有任务执行完成
    wg.Wait()
    fmt.Println("All tasks completed")
}

// 1. 订单超时取消
func OrderTimeoutCancel(orderID string) {
    mtimewheel.After(30*time.Minute, func() {
        // 取消超时未支付的订单
        cancelOrder(orderID)
    })
}

// 2. 缓存过期
func SetCacheWithTTL(key string, value interface{}, ttl time.Duration) {
    // 设置缓存
    cache.Set(key, value)

    // 延迟删除
    mtimewheel.After(ttl, func() {
        cache.Delete(key)
    })
}

// 3. 限流令牌桶补充
type TokenBucket struct {
    tokens int
    capacity int
}

func (tb *TokenBucket) Start() {
    // 每100ms补充一个令牌
    mtimewheel.InitDefault()

    mtimewheel.Default().AddTask("token-refill", 100*time.Millisecond, func(ctx context.Context) error {
        if tb.tokens < tb.capacity {
            tb.tokens++
        }
        return nil
    })
}

// 4. 延迟消息发送
func SendDelayedMessage(userID string, message string, delay time.Duration) {
    mtimewheel.After(delay, func() {
        sendNotification(userID, message)
    })
}

// 5. 重试机制
func RetryWithDelay(taskID string, fn func() error, maxRetries int) {
    var attempt int

    var execute func()
    execute = func() {
        attempt++
        if err := fn(); err != nil {
            if attempt < maxRetries {
                // 指数退避
                delay := time.Duration(attempt) * time.Second
                mtimewheel.After(delay, execute)
            }
        }
    }

    execute()
}


========== 完整示例：订单系统 ==========

package main

import (
    "admin_server/component/mlog"
    "admin_server/component/mtimewheel"
    "context"
    "fmt"
    "time"
    "go.uber.org/zap"
)

type Order struct {
    ID        string
    UserID    string
    Amount    float64
    Status    string
    CreatedAt time.Time
}

var orders = make(map[string]*Order)

func main() {
    // 初始化日志
    mlog.Init(&mlog.Config{
        Level:   "info",
        Format:  "console",
        Console: true,
    })

    // 创建时间轮
    tw := mtimewheel.New()
    tw.Start()
    defer tw.Stop()

    // 创建订单
    order := createOrder("user-123", 99.99)

    // 设置30分钟超时取消
    tw.AddTask(fmt.Sprintf("order-timeout-%s", order.ID), 30*time.Minute, func(ctx context.Context) error {
        return handleOrderTimeout(order.ID)
    })

    // 设置5分钟后发送提醒
    tw.AddTask(fmt.Sprintf("order-remind-%s", order.ID), 5*time.Minute, func(ctx context.Context) error {
        return sendPaymentReminder(order.ID)
    })

    mlog.Info("Order created with timeout",
        zap.String("order_id", order.ID),
        zap.Float64("amount", order.Amount),
    )

    // 模拟订单支付
    time.Sleep(10 * time.Second)
    payOrder(order.ID)

    // 防止退出
    time.Sleep(1 * time.Hour)
}

func createOrder(userID string, amount float64) *Order {
    order := &Order{
        ID:        fmt.Sprintf("ORDER-%d", time.Now().Unix()),
        UserID:    userID,
        Amount:    amount,
        Status:    "pending",
        CreatedAt: time.Now(),
    }
    orders[order.ID] = order
    return order
}

func handleOrderTimeout(orderID string) error {
    order, ok := orders[orderID]
    if !ok {
        return fmt.Errorf("order not found: %s", orderID)
    }

    if order.Status == "pending" {
        order.Status = "cancelled"
        mlog.Info("Order cancelled due to timeout",
            zap.String("order_id", orderID),
        )
        // 释放库存、发送通知等...
    }

    return nil
}

func sendPaymentReminder(orderID string) error {
    order, ok := orders[orderID]
    if !ok {
        return fmt.Errorf("order not found: %s", orderID)
    }

    if order.Status == "pending" {
        mlog.Info("Sending payment reminder",
            zap.String("order_id", orderID),
            zap.String("user_id", order.UserID),
        )
        // 发送短信/邮件提醒...
    }

    return nil
}

func payOrder(orderID string) {
    order, ok := orders[orderID]
    if !ok {
        return
    }

    order.Status = "paid"
    mlog.Info("Order paid",
        zap.String("order_id", orderID),
    )

    // 设置发货超时检查（24小时）
    mtimewheel.Default().AddTask(
        fmt.Sprintf("delivery-timeout-%s", orderID),
        24*time.Hour,
        func(ctx context.Context) error {
            return handleDeliveryTimeout(orderID)
        },
    )
}

func handleDeliveryTimeout(orderID string) error {
    mlog.Warn("Order delivery timeout",
        zap.String("order_id", orderID),
    )
    // 提醒商家发货...
    return nil
}


========== 注意事项 ==========

1. 时间轮的最大延迟 = tick * wheelSize
   例如：tick=100ms, wheelSize=100，最大延迟为 10s

2. 如果需要更大范围的延迟，请使用分层时间轮（HierarchicalTimeWheel）

3. 任务回调函数中的 panic 会被自动捕获，不会影响其他任务

4. 建议在生产环境使用分层时间轮，支持从毫秒到天的延迟

5. 任务执行时间过长可能会影响精度，建议将耗时操作异步处理

6. 时间轮是单线程调度的，适合大量轻量级延迟任务

7. 停止时间轮时，未执行的任务会被丢弃，请确保在停止前处理重要任务

8. 使用 Default() 获取的全局时间轮是单例的，应用生命周期内只需 InitDefault() 一次
