========== 任务管理 ==========

func main() {
    cron := mcron.New()
    cron.Start()
    defer cron.Stop()

    // 添加任务
    id, err := cron.AddFunc("removable-job", mcron.Every(time.Minute), func(ctx context.Context) error {
        fmt.Println("This job will be removed")
        return nil
    })
    if err != nil {
        panic(err)
    }

    // 立即执行一次（阻塞）
    err = cron.RunOnce(id)
    if err != nil {
        fmt.Printf("RunOnce failed: %v\n", err)
    }

    // 立即异步执行一次
    cron.RunOnceAsync(id)

    // 获取任务信息
    entry := cron.GetEntry(id)
    fmt.Printf("Job: %s, RunCount: %d, ErrorCount: %d\n",
        entry.Name, entry.RunCount, entry.ErrorCount)

    // 列出所有任务
    entries := cron.ListEntries()
    for _, e := range entries {
        fmt.Printf("Job: %s, NextRun: %v\n", e.Name, e.NextRun)
    }

    // 移除任务
    time.Sleep(5 * time.Second)
    cron.Remove(id)
}

package main

import (
    "admin_server/component/mcron"
    "admin_server/component/mlog"
    "context"
    "time"
    "go.uber.org/zap"
)

func main() {
    // 初始化日志
    mlog.Init(&mlog.Config{
        Level:   "info",
        Format:  "console",
        Console: true,
    })

    // 创建调度器
    cron := mcron.New()
    cron.Start()
    defer cron.Stop()

    // 1. 每分钟执行一次的心跳任务
    cron.AddFunc("heartbeat", mcron.Every(time.Minute), func(ctx context.Context) error {
        mlog.Info("Heartbeat check", zap.String("status", "alive"))
        return nil
    })

    // 2. 每小时执行一次的数据备份
    cron.AddFunc("data-backup", mcron.PredefinedSchedules.EveryHour, func(ctx context.Context) error {
        mlog.Info("Starting data backup...")
        // 执行备份逻辑
        // ...
        mlog.Info("Data backup completed")
        return nil
    })

    // 3. 每天凌晨2点执行的数据清理
    cron.AddFunc("data-cleanup", mcron.Cron("0 0 2 * * *"), func(ctx context.Context) error {
        mlog.Info("Starting data cleanup...")
        // 执行清理逻辑
        // ...
        mlog.Info("Data cleanup completed")
        return nil
    })

    // 4. 每周一早上8点发送周报
    cron.AddFunc("weekly-report", mcron.Cron("0 0 8 * * 1"), func(ctx context.Context) error {
        mlog.Info("Sending weekly report...")
        // 发送邮件逻辑
        // ...
        mlog.Info("Weekly report sent")
        return nil
    })

    // 5. 延迟10秒后执行的一次性任务
    cron.AddFunc("delayed-task",
        mcron.Delay(10*time.Second, mcron.Every(time.Hour)),
        func(ctx context.Context) error {
            mlog.Info("Delayed task executed")
            return nil
        })

    // 打印所有任务
    entries := cron.ListEntries()
    for _, entry := range entries {
        mlog.Info("Scheduled job",
            zap.String("name", entry.Name),
            zap.Time("next_run", entry.NextRun),
        )
    }

    // 保持运行
    select {}
}


========== 注意事项 ==========

1. 任务应该是幂等的，因为可能由于网络等原因导致重复执行
2. 长时间运行的任务应该设置超时，避免阻塞调度器
3. 使用 context 进行任务取消和超时控制
4. 任务中的 panic 会被自动捕获并记录，不会影响其他任务
5. 建议在应用关闭时调用 Stop() 优雅停止调度器
6. 任务执行时间过长可能会影响下一次调度（对于固定间隔的任务）
